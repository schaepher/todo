{{define "todo-toast"}} class TodoToast { constructor(containerId) { this.container = document.getElementById(containerId); } show(message, type) { const el = document.createElement('div'); el.className = 'toast toast-' + type; el.textContent = message; this.container.appendChild(el); setTimeout(() => { if (el.parentNode) el.remove(); }, 3000); } success(msg) { this.show(msg, 'success'); } error(msg) { this.show(msg, 'error'); } } {{end}}
